## 用法

```ts
const ad = useAd() // 同步句柄——无 await、无 res.ok

if (ad.ok) {
  const rv = ad.data.rewardedVideo('adunit-xxxx') // 同 id 复用实例
  rv.onClose((res) => {
    if (res.isEnded) console.log('观看完毕，发放奖励')
  })
  await rv.load()
  await rv.show()
} else if (ad.error.code === 'ad.unsupported') {
  // Web 无广告联盟标准 API → 降级路径（宿主桥）
}
```
