## 用法

```ts
const a = useAudio('https://cdn.example.com/bgm.mp3') // 同步句柄——无 await、无 res.ok

if (a.ok) {
  a.data.setVolume(0.8)
  a.data.setLoop(true)
  await a.data.play()
  a.data.on('ended', () => console.log('播放结束'))
  // a.data.destroy() 释放资源
} else if (a.error.code === 'audio.unsupported') {
  // 桥未提供音频 API → 降级路径
}
```
