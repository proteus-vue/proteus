# useTranslation

## 用法

```ts
const tr = useTranslation() // 同步句柄——无 await、无 res.ok

if (tr.ok) {
  tr.data.onTrigger((res) => console.log("用户触发翻译:", res.locale, res.type))
  tr.data.onOff(() => console.log("用户取消翻译"))
} else if (tr.error.code === 'translation.unsupported') {
  // 平台不支持 → 降级路径
}
```
